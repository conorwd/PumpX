 1 | # pump_fun_py
 2 | 
 3 | Python library to trade on pump.fun. 
 4 | 
 5 | Updated: 10/30/2024
 6 | 
 7 | Clone the repo, and add your Private Key (Base58 string) and RPC to the config.py.
 8 | 
 9 | If you are using the new version of solana-py, use **send_legacy_transaction()** instead of **send_transaction()**. 
10 |

17 | I am not your personal tech support. READ THE FAQS. 
18 | 
19 | Telegram: @AL_THE_BOT_FATHER
20 | 
21 | ### FAQS
22 | 
23 | **What format should my private key be in?** 
24 | 
25 | The private key should be in the base58 string format, not bytes. 
26 | 
27 | **Why are my transactions being dropped?** 
28 | 
29 | You get what you pay for. Don't use the main-net RPC, just spend the money for Helius or Quick Node.
30 | 
31 | **How do I change the fee?** 
32 | 
33 | Modify the UNIT_BUDGET and UNIT_PRICE in the config.py. 
34 | 
35 | **Does this code work on devnet?**
36 | 
37 | No. 
38 | 
39 | ### Example
40 | 
41 | ```
42 | from pump_fun import buy
43 | 
44 | # Buy Example
45 | mint_str = "pump_token_address"
46 | sol_in = .1
47 | slippage = 25
48 | buy(mint_str, sol_in, slippage)
49 | ```
50 | 
51 | ```
52 | from pump_fun import sell
53 | 
54 | # Sell Example
55 | mint_str = "pump_token_address"
56 | percentage = 100
57 | slippage = 25
58 | sell(mint_str, percentage, slippage)
59 | ```

/pump_fun_py/coin_data.py:
----------------------------------------------------------------------------------------------------
 1 | from solders.pubkey import Pubkey  # type: ignore
 2 | from spl.token.instructions import get_associated_token_address
 3 | from construct import Padding, Struct, Int64ul, Flag
 4 | from config import client
 5 | from constants import PUMP_FUN_PROGRAM
 6 | 
 7 | def get_virtual_reserves(bonding_curve: Pubkey):
 8 |     bonding_curve_struct = Struct(
 9 |         Padding(8),
10 |         "virtualTokenReserves" / Int64ul,
11 |         "virtualSolReserves" / Int64ul,
12 |         "realTokenReserves" / Int64ul,
13 |         "realSolReserves" / Int64ul,
14 |         "tokenTotalSupply" / Int64ul,
15 |         "complete" / Flag
16 |     )
17 |     
18 |     try:
19 |         account_info = client.get_account_info(bonding_curve)
20 |         data = account_info.value.data
21 |         parsed_data = bonding_curve_struct.parse(data)
22 |         return parsed_data
23 |     except Exception:
24 |         return None
25 | 
26 | def derive_bonding_curve_accounts(mint_str: str):
27 |     try:
28 |         mint = Pubkey.from_string(mint_str)
29 |         bonding_curve, _ = Pubkey.find_program_address(
30 |             ["bonding-curve".encode(), bytes(mint)],
31 |             PUMP_FUN_PROGRAM
32 |         )
33 |         associated_bonding_curve = get_associated_token_address(bonding_curve, mint)
34 |         return bonding_curve, associated_bonding_curve
35 |     except Exception:
36 |         return None, None
37 | 
38 | def get_coin_data(mint_str: str):
39 |     bonding_curve, associated_bonding_curve = derive_bonding_curve_accounts(mint_str)
40 |     if bonding_curve is None or associated_bonding_curve is None:
41 |         return None
42 | 
43 |     virtual_reserves = get_virtual_reserves(bonding_curve)
44 |     if virtual_reserves is None:
45 |         return None
46 | 
47 |     try:
48 |         virtual_token_reserves = int(virtual_reserves.virtualTokenReserves)
49 |         virtual_sol_reserves = int(virtual_reserves.virtualSolReserves)
50 |         token_total_supply = int(virtual_reserves.tokenTotalSupply)
51 |         complete = bool(virtual_reserves.complete)
52 |         
53 |         return {
54 |             "mint": mint_str,
55 |             "bonding_curve": str(bonding_curve),
56 |             "associated_bonding_curve": str(associated_bonding_curve),
57 |             "virtual_token_reserves": virtual_token_reserves,
58 |             "virtual_sol_reserves": virtual_sol_reserves,
59 |             "token_total_supply": token_total_supply,
60 |             "complete": complete
61 |         }
62 |     except Exception:
63 |         return None
